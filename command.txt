# =============================================================================
# StableFlow Experiment Commands  — Quick Run (~1 hour, CPU-offload mode)
# Run all commands from the repo root: e:/Cherry_on_top
# Set HF_TOKEN to your Hugging Face token before running.
#
# Time budget breakdown (cpu_offload, ~20-30s per image at 4 steps):
#   1b. Vitality sweep  :  4 refs + 57×4 ablated  = 232 images  ~35 min
#   2a. Semantic sweep  :  14 full (pre-cached) + 19×7×2 ablated =  ~15 min
#   3.  Attention maps  :  1 prompt, 4 steps                      ~3 min
#   4.  Attn balance    :  2 prompts, 4 steps                     ~5 min
#   Plots               :  negligible
#   Total                                                         ~58 min
#
# Key speed tricks applied in the scripts:
#   - Reference / full-model images are pre-generated ONCE before the layer
#     loop (not regenerated 57× inside it).
#   - n_steps=4  (down from 15)
#   - n_prompts=4, n_seeds=1 for vitality
#   - n_pairs=1, --only_mm for semantic (19 layers instead of 57)
#   - prompts_n=2 for attention balance
# =============================================================================

# ---------------------------------------------------------------------------
# 0. Environment setup  (run once)
# ---------------------------------------------------------------------------

conda env create -f environment.yml
conda activate stable-flow
pip install -e src/diffusers
pip install transformers accelerate torchvision matplotlib seaborn openai

# ---------------------------------------------------------------------------
# 1a. Generate prompts  (no GPU, ~1s)
# ---------------------------------------------------------------------------

python generate_prompts.py

# Output:
#   prompts/vitality_prompts.json
#   prompts/semantic_prompts.json

# ---------------------------------------------------------------------------
# 1b. Vitality sweep  (~35 min)
#   232 images: 4 refs (pre-cached) + 57 layers × 4 prompts × 1 seed
#   Saves incrementally — safe to interrupt and resume.
# ---------------------------------------------------------------------------

python experiments/vitality_sweep.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --n_prompts 4 \
    --n_seeds 1 \
    --n_steps 4 \
    --device cuda \
    --cpu_offload

# Output: results/vitality_scores.json

# ---------------------------------------------------------------------------
# 1c. Plot vitality  (~1s)
# ---------------------------------------------------------------------------

python experiments/plot_vitality.py \
    --scores results/vitality_scores.json \
    --out results/vitality_plot.png \
    --threshold 0.92

# Output: results/vitality_plot.png

# ---------------------------------------------------------------------------
# 2a. Semantic sensitivity sweep  (~15 min)
#   14 full-model images pre-cached, then 19 MM layers × 7 categories × 2 ablated
#   --only_mm skips the 38 single-stream blocks for speed.
# ---------------------------------------------------------------------------

python experiments/semantic_sensitivity.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --n_pairs 1 \
    --n_steps 4 \
    --only_mm \
    --device cuda \
    --cpu_offload

# Output: results/semantic_sensitivity.npy + results/semantic_sensitivity.json

# ---------------------------------------------------------------------------
# 2b. Plot semantic heatmap  (~1s)
# ---------------------------------------------------------------------------

python experiments/plot_semantic_heatmap.py \
    --sensitivity results/semantic_sensitivity.npy \
    --vitality results/vitality_scores.json \
    --threshold 0.92

# Outputs:
#   results/semantic_heatmap.png
#   results/vitality_semantic_combined.png

# ---------------------------------------------------------------------------
# 3. Attention maps  (~3 min)
# ---------------------------------------------------------------------------

python experiments/attention_maps.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --prompt "a red wooden chair on a stone floor" \
    --tokens "red,wooden,chair,stone,floor" \
    --seed 42 \
    --n_steps 4 \
    --device cuda \
    --fixed_layer 5 \
    --cpu_offload

# Outputs:
#   results/attention_source_image.png
#   results/attention_maps_per_layer.png
#   results/attention_temporal.png
#   results/attention_data.pt

# ---------------------------------------------------------------------------
# 4. Attention balance  (~5 min)
# ---------------------------------------------------------------------------

python experiments/attention_balance.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --prompts_n 2 \
    --vitality results/vitality_scores.json \
    --n_steps 4 \
    --device cuda \
    --cpu_offload

# Outputs:
#   results/attention_balance.json
#   results/vitality_attention_correlation.png

# ---------------------------------------------------------------------------
# 4b. Regenerate correlation plot only  (~1s)
# ---------------------------------------------------------------------------

python experiments/plot_correlation.py \
    --vitality results/vitality_scores.json \
    --balance results/attention_balance.json \
    --out results/vitality_attention_correlation.png \
    --threshold 0.92

# ---------------------------------------------------------------------------
# Full pipeline  (copy-paste to run end-to-end, ~1 hour)
# ---------------------------------------------------------------------------
