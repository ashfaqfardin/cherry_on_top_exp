# =============================================================================
# StableFlow Experiment Commands  (CPU-offload mode)
# Run all commands from the repo root: e:/Cherry_on_top  (or stable-flow/)
# Set HF_TOKEN to your Hugging Face token before running.
#
# --cpu_offload uses enable_sequential_cpu_offload(): model layers are kept in
# RAM and moved to GPU one at a time, so you only need ~6-8 GB VRAM instead
# of ~24 GB, at the cost of significantly slower inference.
# =============================================================================

# ---------------------------------------------------------------------------
# 0. Environment setup  (run once)
# ---------------------------------------------------------------------------

conda env create -f environment.yml
conda activate stable-flow

# Install the custom FLUX diffusers fork in editable mode
pip install -e src/diffusers

# Install additional experiment dependencies
pip install transformers accelerate torchvision matplotlib seaborn openai

# ---------------------------------------------------------------------------
# 1a. Generate prompts  (no GPU needed)
# ---------------------------------------------------------------------------

# Option A: use built-in hardcoded prompts (no API key needed)
python generate_prompts.py

# Option B: generate via OpenAI GPT-4o-mini
python generate_prompts.py --openai_key YOUR_OPENAI_KEY

# Output:
#   prompts/vitality_prompts.json     (64 diverse prompts)
#   prompts/semantic_prompts.json     (7 categories × 50 contrastive pairs)

# ---------------------------------------------------------------------------
# 1b. Vitality sweep  (saves incrementally — safe to interrupt and resume)
# ---------------------------------------------------------------------------

python experiments/vitality_sweep.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --n_steps 15 \
    --device cuda \
    --cpu_offload

# Output: results/vitality_scores.json

# ---------------------------------------------------------------------------
# 1c. Plot vitality bar chart  (no GPU needed)
# ---------------------------------------------------------------------------

python experiments/plot_vitality.py \
    --scores results/vitality_scores.json \
    --out results/vitality_plot.png \
    --threshold 0.92

# Output: results/vitality_plot.png

# ---------------------------------------------------------------------------
# 2a. Semantic sensitivity sweep
# ---------------------------------------------------------------------------

python experiments/semantic_sensitivity.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --n_pairs 10 \
    --n_steps 15 \
    --device cuda \
    --cpu_offload

# Use --n_pairs 5 for a faster run (fewer contrastive pairs per category).
# Output: results/semantic_sensitivity.npy  +  results/semantic_sensitivity.json

# ---------------------------------------------------------------------------
# 2b. Plot semantic heatmap  (no GPU needed)
# ---------------------------------------------------------------------------

python experiments/plot_semantic_heatmap.py \
    --sensitivity results/semantic_sensitivity.npy \
    --vitality results/vitality_scores.json \
    --threshold 0.92

# Outputs:
#   results/semantic_heatmap.png
#   results/vitality_semantic_combined.png

# ---------------------------------------------------------------------------
# 3. Attention maps
# ---------------------------------------------------------------------------

python experiments/attention_maps.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --prompt "a red wooden chair on a stone floor" \
    --tokens "red,wooden,chair,stone,floor" \
    --seed 42 \
    --n_steps 15 \
    --device cuda \
    --fixed_layer 5 \
    --cpu_offload

# Outputs:
#   results/attention_source_image.png
#   results/attention_maps_per_layer.png
#   results/attention_temporal.png
#   results/attention_data.pt

# ---------------------------------------------------------------------------
# 4. Attention balance
# ---------------------------------------------------------------------------

python experiments/attention_balance.py \
    --model_path black-forest-labs/FLUX.1-dev \
    --hf_token $HF_TOKEN \
    --prompts_n 8 \
    --vitality results/vitality_scores.json \
    --n_steps 15 \
    --device cuda \
    --cpu_offload

# Outputs:
#   results/attention_balance.json
#   results/vitality_attention_correlation.png

# ---------------------------------------------------------------------------
# 4b. Regenerate correlation plot from saved results  (no GPU needed)
# ---------------------------------------------------------------------------

python experiments/plot_correlation.py \
    --vitality results/vitality_scores.json \
    --balance results/attention_balance.json \
    --out results/vitality_attention_correlation.png \
    --threshold 0.92

# ---------------------------------------------------------------------------
# Full pipeline  (copy-paste to run end-to-end)
# ---------------------------------------------------------------------------

export HF_TOKEN=your_hf_token_here

python generate_prompts.py

python experiments/vitality_sweep.py \
    --model_path black-forest-labs/FLUX.1-dev --hf_token $HF_TOKEN \
    --n_steps 15 --device cuda --cpu_offload

python experiments/plot_vitality.py

python experiments/semantic_sensitivity.py \
    --model_path black-forest-labs/FLUX.1-dev --hf_token $HF_TOKEN \
    --n_pairs 10 --n_steps 15 --device cuda --cpu_offload

python experiments/plot_semantic_heatmap.py

python experiments/attention_maps.py \
    --model_path black-forest-labs/FLUX.1-dev --hf_token $HF_TOKEN \
    --prompt "a red wooden chair on a stone floor" \
    --tokens "red,wooden,chair,stone,floor" \
    --seed 42 --n_steps 15 --device cuda --fixed_layer 5 --cpu_offload

python experiments/attention_balance.py \
    --model_path black-forest-labs/FLUX.1-dev --hf_token $HF_TOKEN \
    --prompts_n 8 --n_steps 15 --device cuda --cpu_offload

python experiments/plot_correlation.py
